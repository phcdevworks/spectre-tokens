How It Works

<html data-spectre-theme="system">   <!-- follows prefers-color-scheme, no script -->
<section data-spectre-theme="dark">   <!-- this section and its children -->
<div data-spectre-theme="high-contrast">   <!-- AAA (7:1) text everywhere inside -->
